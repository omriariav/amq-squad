// Probe: does execve("/dev/fd/N") of a held regular-file descriptor work on
// this macOS host, for a native binary and for node with a script argument?
package main

import (
	"fmt"
	"os"
	"syscall"

	"golang.org/x/sys/unix"
)

func main() {
	target := os.Args[1]
	fd, err := unix.Open(target, unix.O_RDONLY|unix.O_CLOEXEC, 0)
	if err != nil {
		panic(err)
	}
	// Clear close-on-exec on the descriptor we exec through.
	if _, err := unix.FcntlInt(uintptr(fd), unix.F_SETFD, 0); err != nil {
		panic(err)
	}
	path := fmt.Sprintf("/dev/fd/%d", fd)
	argv := append([]string{path}, os.Args[2:]...)
	err = syscall.Exec(path, argv, os.Environ())
	fmt.Println("exec failed:", err)
	os.Exit(1)
}
