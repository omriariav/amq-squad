package seatcommit

import (
	"errors"
	"os"
	"testing"

	"github.com/omriariav/amq-squad/v3/internal/sealchain"
)

// This source is intentionally specific to v3.2.0-rc.1, where
// QuarantineDir is part of the package contract.
func TestCompatProbe(t *testing.T) {
	root := os.Getenv("AMQSQ_0E_FIXTURE_ROOT")
	ledger, err := ReadLedger(root, "qa")
	t.Logf("commit error: %v", err)
	t.Logf("commit tampered: %v", errors.Is(err, sealchain.ErrTampered))
	t.Logf("commit handle: %s", ledger.Handle)
	entries, qerr := os.ReadDir(QuarantineDir(root))
	if qerr != nil {
		t.Logf("commit quarantine: %v", qerr)
		return
	}
	for _, entry := range entries {
		t.Logf("commit quarantine: %s", entry.Name())
	}
}
